    \ifnum\i=#1
      \node[draw=acc,fill=acc,text=white,rounded corners=1.5pt,minimum width=3.6cm,minimum height=4.2mm,font=\scriptsize\bfseries] (s\i) at (\i*3.95,0) {\i. \name};
    \else
      \node[draw=acc!60,fill=soft,text=acc,rounded corners=1.5pt,minimum width=3.6cm,minimum height=4.2mm,font=\scriptsize] (s\i) at (\i*3.95,0) {\i. \name};
    \fi
  }
  \foreach \i [evaluate=\i as \j using int(\i+1)] in {1,...,6}{\draw[-{Stealth[length=3pt]},acc!70] (s\i.east) -- (s\j.west);}
  \end{tikzpicture}}
